   123→async def handle_password(update: Update, context: ContextTypes.DEFAULT_TYPE):
   124→    """معالجة إدخال كلمة السر"""
   125→    user = update.effective_user
   126→    text = update.message.text.strip()
   127→
   128→    # المدير لا يحتاج كلمة سر
   129→    if is_admin(user.id):
   130→        await handle_message(update, context)
   131→        return
   132→
   133→    # إذا كان المستخدم ينتظر كلمة السر
   134→    if user.id in waiting_for_password:
   135→        db.log_join_action(user.id, "password_attempt", text)
   136→
   137→        if text == JOIN_PASSWORD:
   138→            # كلمة السر صحيحة
   139→            waiting_for_password.discard(user.id)
   140→            db.approve_user(user.id)
   141→            ai.load_history_from_db(user.id)
   142→            await update.message.reply_text(JOIN_SUCCESS_MESSAGE, parse_mode="Markdown")
   143→
   144→            # إعلام المدير
   145→            for admin_id in ADMIN_IDS:
   146→                try:
   147→                    username = f"@{user.username}" if user.username else "بدون معرف"
   148→                    await context.bot.send_message(
   149→                        chat_id=admin_id,
   150→                        text=(
   151→                            f"✅ **عضو جديد انضم!**\n\n"
   152→                            f"👤 {user.first_name} {username}\n"
   153→                            f"🆔 `{user.id}`"
   154→                        ),
   155→                        parse_mode="Markdown"
   156→                    )
   157→                except Exception:
   158→                    pass
   159→        else:
   160→            # كلمة السر خاطئة
   161→            db.increment_join_attempt(user.id)
   162→            await update.message.reply_text(JOIN_FAIL_MESSAGE, parse_mode="Markdown")
   163→
   164→            # إعلام المدير بمحاولة خاطئة
   165→            for admin_id in ADMIN_IDS:
   166→                try:
   167→                    username = f"@{user.username}" if user.username else "بدون معرف"
   168→                    await context.bot.send_message(
   169→                        chat_id=admin_id,
   170→                        text=(
   171→                            f"⚠️ **محاولة كلمة سر خاطئة!**\n\n"
   172→                            f"👤 {user.first_name} {username}\n"
   173→                            f"🆔 `{user.id}`\n"
   174→                            f"🔑 المحاولة: `{text}`"
   175→                        ),
   176→                        parse_mode="Markdown"
   177→                    )
   178→                except Exception:
   179→                    pass
   180→        return
   181→
   182→    # إذا لم يكن موافقاً عليه ولم يكن ينتظر كلمة السر
   183→    if not db.is_approved(user.id) and not is_admin(user.id):
   184→        waiting_for_password.add(user.id)
   185→        await update.message.reply_text(JOIN_REQUEST_MESSAGE, parse_mode="Markdown")
   186→        return
   187→
   188→    # مستخدم عادي موافق عليه - معالجة الرسالة
   189→    await handle_message(update, context)

   192→async def handle_message(update: Update, context: ContextTypes.DEFAULT_TYPE):
   193→    """معالجة الرسائل العادية - إرسالها للذكاء الاصطناعي والرد"""
   194→    user = update.effective_user
   195→    user_message = update.message.text
   196→
   197→    # التحقق من الحظر
   198→    if db.is_blocked(user.id):
   199→        await update.message.reply_text("⛔ تم حظرك من استخدام هذا البوت.")
   200→        return
   201→
   202→    # التحقق من الموافقة
   203→    if not db.is_approved(user.id) and not is_admin(user.id):
   204→        waiting_for_password.add(user.id)
   205→        await update.message.reply_text(JOIN_REQUEST_MESSAGE, parse_mode="Markdown")
   206→        return
   207→
   208→    # تسجيل/تحديث المستخدم
   209→    db.register_user(
   210→        user_id=user.id,
   211→        username=user.username,
   212→        first_name=user.first_name,
   213→        last_name=user.last_name,
   214→        language_code=user.language_code
   215→    )
   216→
   217→    logger.info(f"رسالة من {user.first_name} (ID: {user.id}): {user_message[:50]}...")
   218→
   219→    # إظهار حالة الكتابة
   220→    await update.message.chat.send_action("typing")
   221→
   222→    try:
   223→        reply = await ai.chat(user.id, user_message)
   224→
   225→        if len(reply) <= 4096:
   226→            await update.message.reply_text(reply)
   227→        else:
   228→            parts = [reply[i:i+4096] for i in range(0, len(reply), 4096)]
   229→            for part in parts:
   230→                await update.message.reply_text(part)
   231→                await asyncio.sleep(0.5)
   232→
   233→    except Exception as e:
   234→        logger.error(f"خطأ في معالجة الرسالة: {e}")
   235→        await update.message.reply_text(
   236→            "❌ حدث خطأ أثناء معالجة رسالتك. حاول مرة أخرى."
   237→        )

   446→async def _show_user_chat(query, user_id: int):
   447→    """عرض محادثة مستخدم"""
   448→    messages = db.get_user_messages(user_id, limit=30)
   449→    user = db.get_user(user_id)
   450→
   451→    if not messages:
   452→        await query.message.reply_text("❌ لا توجد رسائل.")
   453→        return
   454→
   455→    name = user['first_name'] if user else str(user_id)
   456→    text = f"💬 **محادثة {name}:**\n━━━━━━━━━━━━━━━━━\n\n"
   457→
   458→    messages.reverse()
   459→
   460→    for msg in messages[:30]:
   461→        role_icon = "👤" if msg['role'] == "user" else "🤖"
   462→        role_name = "المستخدم" if msg['role'] == "user" else "البوت"
   463→        content = msg['content'][:200]
   464→        if len(msg['content']) > 200:
   465→            content += "..."
   466→        time = msg['timestamp'][:16]
   467→        text += f"{role_icon} **{role_name}** ({time}):\n{content}\n\n"
   468→
   469→    if len(text) > 4096:
   470→        parts = [text[i:i+4096] for i in range(0, len(text), 4096)]
   471→        for part in parts:
   472→            await query.message.reply_text(part, parse_mode="Markdown")
   473→            await asyncio.sleep(0.5)
   474→    else:
   475→        await query.message.reply_text(text, parse_mode="Markdown")
