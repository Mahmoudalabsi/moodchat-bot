   122→async def handle_message(update: Update, context: ContextTypes.DEFAULT_TYPE):
   123→    """معالجة الرسائل العادية - إرسالها للذكاء الاصطناعي والرد"""
   124→    user = update.effective_user
   125→    user_message = update.message.text
   126→
   127→    # التحقق من حظر المستخدم
   128→    if db.is_blocked(user.id):
   129→        await update.message.reply_text("⛔ تم حظرك من استخدام البوت.")
   130→        return
   131→
   132→    # التحقق من الحد اليومي
   133→    if DAILY_MESSAGE_LIMIT > 0:
   134→        today_msgs = db.get_messages_today()  # هذه للكل، نحتاج لكل مستخدم
   135→        user_msg_count = len([m for m in db.get_user_messages(user.id, limit=1000)
   136→                             if m["timestamp"].startswith(datetime.now().strftime("%Y-%m-%d"))])
   137→        if user_msg_count >= DAILY_MESSAGE_LIMIT:
   138→            await update.message.reply_text(
   139→                f"⚠️ وصلت للحد اليومي ({DAILY_MESSAGE_LIMIT} رسالة). حاول غداً!"
   140→            )
   141→            return
   142→
   143→    # تسجيل/تحديث المستخدم
   144→    db.register_user(
   145→        user_id=user.id,
   146→        username=user.username,
   147→        first_name=user.first_name,
   148→        last_name=user.last_name,
   149→        language_code=user.language_code
   150→    )
   151→
   152→    logger.info(f"رسالة من {user.first_name} (ID: {user.id}): {user_message[:50]}...")
   153→
   154→    # إظهار حالة الكتابة
   155→    await update.message.chat.send_action("typing")
   156→
   157→    try:
   158→        # إرسال الرسالة للذكاء الاصطناعي
   159→        reply = await ai.chat(user.id, user_message)
   160→
   161→        # تقسيم الرسالة إذا كانت طويلة جداً
   162→        if len(reply) <= 4096:
   163→            await update.message.reply_text(reply)
   164→        else:
   165→            parts = [reply[i:i+4096] for i in range(0, len(reply), 4096)]
   166→            for part in parts:
   167→                await update.message.reply_text(part)
   168→                await asyncio.sleep(0.5)
   169→
   170→    except Exception as e:
   171→        logger.error(f"خطأ في معالجة الرسالة: {e}")
   172→        await update.message.reply_text(
   173→            "❌ حدث خطأ أثناء معالجة رسالتك. حاول مرة أخرى."
   174→        )

   256→    """معالجة أزرار لوحة التحكم"""

   276→        )

   278→        await _export_data(query)

   280→        await query.message.reply_text(
   281→            "🚫 أوامر الحظر:\n"
   282→            "/block <user_id> - حظر مستخدم\n"

   284→        )
   285→    elif data.startswith("user_detail_"):
   286→        user_id = int(data.replace("user_detail_", ""))

   381→    """عرض محادثة مستخدم معين"""
   382→    messages = db.get_user_messages(user_id, limit=30)
   383→    user = db.get_user(user_id)
   384→
   385→    if not messages:
   386→        await query.message.reply_text("❌ لا توجد رسائل لهذا المستخدم.")
   387→        return
   388→
   389→    name = user['first_name'] if user else str(user_id)
   390→    text = f"💬 **محادثة {name}:**\n━━━━━━━━━━━━━━━━━\n\n"
   391→
   392→    # عكس الرسائل لعرضها بالترتيب الزمني
   393→    messages.reverse()
   394→
   395→    for msg in messages[:30]:
   396→        role_icon = "👤" if msg['role'] == "user" else "🤖"
   397→        role_name = "المستخدم" if msg['role'] == "user" else "البوت"
   398→        content = msg['content'][:200]  # اقتطاع الرسائل الطويلة
   399→        if len(msg['content']) > 200:
   400→            content += "..."
   401→        time = msg['timestamp'][:16]
   402→        text += f"{role_icon} **{role_name}** ({time}):\n{content}\n\n"
   403→
   404→    if len(text) > 4096:
   405→        # تقسيم الرسالة
   406→        parts = [text[i:i+4096] for i in range(0, len(text), 4096)]
   407→        for part in parts:
   408→            await query.message.reply_text(part, parse_mode="Markdown")
   409→            await asyncio.sleep(0.5)
   410→    else:
   411→        await query.message.reply_text(text, parse_mode="Markdown")
